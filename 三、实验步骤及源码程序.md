```python
"""
数据预处理模块
包含数据读取、标签编码和Z-Score标准化
"""

import pandas as pd
import numpy as np
import os
from config import DATASET_PATH

class DataPreprocessor:
    """数据预处理类"""
    
    def __init__(self):
        self.label_mapping = {}
        self.mean = None
        self.std = None
    
    def load_data(self, file_path=None):
        """
        读取CSV数据集
        
        Returns:
            tuple: (features, labels) - 特征矩阵和标签向量
        """
        if file_path is None:
            file_path = DATASET_PATH
            
        # 读取CSV文件
        df = pd.read_csv(file_path)
        print(f"数据集形状: {df.shape}")
        print(f"列名: {list(df.columns)}")
        
        # 分离特征和标签
        features = df.iloc[:, :-1].values  # 除最后一列外的所有特征
        labels = df.iloc[:, -1].values     # 最后一列是类别标签
        
        print(f"特征矩阵形状: {features.shape}")
        print(f"标签向量形状: {labels.shape}")
        print(f"唯一类别: {np.unique(labels)}")
        
        return features, labels
    
    def encode_labels(self, labels):
        """
        将字符型标签编码为数字 (0-6)
        
        Args:
            labels: 原始字符标签向量
            
        Returns:
            tuple: (encoded_labels, label_mapping) - 编码后的标签和映射字典
        """
        unique_labels = np.unique(labels)
        self.label_mapping = {label: idx for idx, label in enumerate(unique_labels)}
        
        # 进行标签编码
        encoded_labels = np.array([self.label_mapping[label] for label in labels])
        
        print("标签映射关系:")
        for label, encoded in self.label_mapping.items():
            print(f"  {label} -> {encoded}")
        
        return encoded_labels, self.label_mapping
    
    def z_score_normalize(self, X):
        """
        Z-Score标准化：x' = (x - μ) / σ
        
        数学原理：
        - μ (mean): 特征的均值
        - σ (std): 特征的标准差
        - 标准化后的数据均值为0，标准差为1
        
        Args:
            X: 原始特征矩阵
            
        Returns:
            tuple: (normalized_X, mean, std) - 标准化后的矩阵、均值和标准差
        """
        # 计算每个特征的均值和标准差
        self.mean = np.mean(X, axis=0)  # 按列计算均值
        self.std = np.std(X, axis=0)    # 按列计算标准差
        
        # 避免除零错误
        self.std = np.where(self.std == 0, 1e-8, self.std)
        
        # 应用Z-Score标准化公式
        X_normalized = (X - self.mean) / self.std
        
        print(f"标准化前特征范围: [{X.min():.2f}, {X.max():.2f}]")
        print(f"标准化后特征范围: [{X_normalized.min():.2f}, {X_normalized.max():.2f}]")
        print(f"标准化后均值: {np.mean(X_normalized, axis=0).mean():.6f}")
        print(f"标准化后标准差: {np.std(X_normalized, axis=0).mean():.6f}")
        
        return X_normalized, self.mean, self.std
    
    def preprocess_pipeline(self):
        """
        完整的数据预处理流程
        
        Returns:
            dict: 包含所有预处理结果的字典
        """
        print("=== 开始数据预处理 ===")
        
        # 1. 加载数据
        print("\n1. 加载数据...")
        X, y = self.load_data()
        
        # 2. 标签编码
        print("\n2. 标签编码...")
        y_encoded, label_mapping = self.encode_labels(y)
        
        # 3. Z-Score标准化
        print("\n3. Z-Score标准化...")
        X_normalized, mean, std = self.z_score_normalize(X)
        
        print("\n=== 数据预处理完成 ===")
        
        return {
            'X_raw': X,
            'y_raw': y,
            'X_normalized': X_normalized,
            'y_encoded': y_encoded,
            'label_mapping': label_mapping,
            'mean': mean,
            'std': std
        }
```

### 1.3 技术说明

**Z-Score标准化公式**：

$$
x' = \frac{x - \mu}{\sigma}
$$

其中：
- $x$ 是原始特征值
- $\mu$ 是该特征的均值
- $\sigma$ 是该特征的标准差
- $x'$ 是标准化后的值

**标准化的必要性**：
1. 消除不同特征间的量纲影响
2. 加速梯度下降类算法的收敛
3. 避免数值计算中的溢出问题
4. 提升算法性能和稳定性

```python
"""
K-Means聚类算法实现
仅使用numpy实现，不依赖sklearn
"""

import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score
import sys
import os
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from config import KMEANS_CONFIG, OUTPUT_DIR, PLOT_PARAMS

# 设置中文字体支持
plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial Unicode MS']
plt.rcParams['axes.unicode_minus'] = False

class MyKMeans:
    """
    K-Means聚类算法实现
    
    算法原理：
    1. 随机初始化k个聚类中心
    2. 分配样本到最近的聚类中心
    3. 更新聚类中心为簇内样本均值
    4. 重复步骤2-3直到收敛
    
    数学公式：
    - 目标函数：J = Σ Σ ||x - μ_k||²
    - 质心更新：μ_k = (1/|C_k|) Σ x_i
    """
    
    def __init__(self, n_clusters=7, max_iters=300, tol=1e-4, random_state=42):
        """
        初始化K-Means参数
        
        Args:
            n_clusters: 聚类数量
            max_iters: 最大迭代次数
            tol: 收敛阈值
            random_state: 随机种子
        """
        self.n_clusters = n_clusters
        self.max_iters = max_iters
        self.tol = tol
        self.random_state = random_state
        self.centers = None
        self.labels = None
        self.inertia = None  # 簇内平方和
        
    def _init_centers(self, X):
        """随机初始化聚类中心"""
        np.random.seed(self.random_state)
        random_indices = np.random.choice(X.shape[0], self.n_clusters, replace=False)
        return X[random_indices]
    
    def _assign_clusters(self, X):
        """将每个样本分配到最近的聚类中心"""
        # 计算每个样本到每个中心的欧氏距离
        distances = np.sqrt(((X - self.centers[:, np.newaxis])**2).sum(axis=2))
        # 返回最近中心的索引
        return np.argmin(distances, axis=0)
    
    def _update_centers(self, X, labels):
        """更新聚类中心为簇内样本的均值"""
        centers = np.zeros((self.n_clusters, X.shape[1]))
        for k in range(self.n_clusters):
            cluster_points = X[labels == k]
            if len(cluster_points) > 0:
                centers[k] = cluster_points.mean(axis=0)
            else:
                # 如果某个簇为空，重新随机初始化
                centers[k] = X[np.random.choice(X.shape[0])]
        return centers
    
    def _calculate_inertia(self, X, labels):
        """计算簇内平方和（衡量聚类紧密度）"""
        inertia = 0
        for k in range(self.n_clusters):
            cluster_points = X[labels == k]
            if len(cluster_points) > 0:
                inertia += np.sum((cluster_points - self.centers[k])**2)
        return inertia
    
    def fit_predict(self, X):
        """训练K-Means模型并返回聚类标签"""
        print(f"开始K-Means聚类，聚类数k={self.n_clusters}")
        
        # 初始化聚类中心
        self.centers = self._init_centers(X)
        
        for iteration in range(self.max_iters):
            # 分配样本到最近的聚类中心
            labels = self._assign_clusters(X)
            
            # 更新聚类中心
            new_centers = self._update_centers(X, labels)
            
            # 检查收敛
            center_shift = np.sqrt(((new_centers - self.centers)**2).sum())
            
            if iteration % 50 == 0:
                inertia = self._calculate_inertia(X, labels)
                print(f"迭代 {iteration}: 中心偏移={center_shift:.6f}, 惯性={inertia:.2f}")
            
            self.centers = new_centers
            
            if center_shift < self.tol:
                print(f"收敛于第 {iteration} 次迭代")
                break
        
        self.labels = labels
        self.inertia = self._calculate_inertia(X, labels)
        
        print(f"K-Means聚类完成！最终惯性: {self.inertia:.2f}")
        return self.labels
    
    def plot_clusters(self, X, y_true=None, save_path=None):
        """可视化聚类结果（使用PCA降维到2D）"""
        # 使用PCA降维到2D
        pca = PCA(n_components=2, random_state=42)
        X_2d = pca.fit_transform(X)
        centers_2d = pca.transform(self.centers)
        
        # 创建图形
        fig, axes = plt.subplots(1, 2 if y_true is not None else 1, figsize=(16, 6))
        
        if y_true is not None:
            # 左图：K-Means聚类结果
            ax1 = axes[0]
        else:
            ax1 = axes if not isinstance(axes, np.ndarray) else axes[0]
        
        scatter1 = ax1.scatter(X_2d[:, 0], X_2d[:, 1], c=self.labels, 
                              cmap='viridis', alpha=0.6, s=30, edgecolors='k', linewidths=0.5)
        ax1.scatter(centers_2d[:, 0], centers_2d[:, 1], c='red', 
                   marker='X', s=300, edgecolors='black', linewidths=2, label='聚类中心')
        ax1.set_title(f'K-Means聚类结果 (k={self.n_clusters})\n惯性: {self.inertia:.2f}', 
                     fontsize=14, fontweight='bold')
        ax1.set_xlabel('第一主成分', fontsize=12)
        ax1.set_ylabel('第二主成分', fontsize=12)
        ax1.legend()
        ax1.grid(True, alpha=0.3)
        
        if y_true is not None:
            # 右图：真实标签
            ax2 = axes[1]
            scatter2 = ax2.scatter(X_2d[:, 0], X_2d[:, 1], c=y_true, 
                                  cmap='tab10', alpha=0.6, s=30, edgecolors='k', linewidths=0.5)
            ax2.set_title('真实标签分布', fontsize=14, fontweight='bold')
            ax2.set_xlabel('第一主成分', fontsize=12)
            ax2.set_ylabel('第二主成分', fontsize=12)
            ax2.grid(True, alpha=0.3)
            
            # 计算ARI和NMI评分
            ari = adjusted_rand_score(y_true, self.labels)
            nmi = normalized_mutual_info_score(y_true, self.labels)
            fig.suptitle(f'ARI: {ari:.4f}, NMI: {nmi:.4f}', 
                        fontsize=16, fontweight='bold', y=0.98)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=PLOT_PARAMS['dpi'], bbox_inches='tight')
            print(f"\n图片已保存到: {save_path}")
        else:
            default_path = os.path.join(OUTPUT_DIR, 'K-Means聚类结果.png')
            plt.savefig(default_path, dpi=PLOT_PARAMS['dpi'], bbox_inches='tight')
            print(f"\n图片已保存到: {default_path}")
        
        plt.show()
        plt.close()
```

#### 2.2.2 DBSCAN聚类算法 (`聚类/dbscan.py`)

```python
"""
DBSCAN聚类算法实现
仅使用numpy实现，不依赖sklearn
"""

import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.metrics import adjusted_rand_score, normalized_mutual_info_score
import sys
import os
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from config import DBSCAN_CONFIG, OUTPUT_DIR, PLOT_PARAMS

# 设置中文字体支持
plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial Unicode MS']
plt.rcParams['axes.unicode_minus'] = False

class MyDBSCAN:
    """
    DBSCAN (Density-Based Spatial Clustering of Applications with Noise) 算法实现
    
    算法原理：
    1. 核心对象：eps邻域内至少包含min_samples个点
    2. 边界点：在核心对象的eps邻域内，但本身不是核心对象
    3. 噪声点：既不是核心对象也不是边界点
    
    关键参数：
    - eps: 邻域半径
    - min_samples: 成为核心对象所需的最小邻居数
    """
    
    def __init__(self, eps=0.5, min_samples=5):
        """
        初始化DBSCAN参数
        
        Args:
            eps: 邻域半径ε
            min_samples: 核心对象的最小邻居数
        """
        self.eps = eps
        self.min_samples = min_samples
        self.labels = None
        self.core_sample_indices = None
        
    def _region_query(self, X, point_idx):
        """查询点point_idx的eps邻域内的所有点"""
        # 计算当前点到所有点的距离
        distances = np.sqrt(np.sum((X - X[point_idx])**2, axis=1))
        # 返回距离小于eps的点的索引
        return np.where(distances < self.eps)[0]
    
    def _expand_cluster(self, X, labels, point_idx, neighbors, cluster_id):
        """
        从核心点扩展簇
        
        Args:
            X: 数据集
            labels: 当前标签数组
            point_idx: 当前核心点索引
            neighbors: 当前核心点的邻域
            cluster_id: 当前簇ID
        """
        # 将当前点标记为cluster_id
        labels[point_idx] = cluster_id
        
        # 使用队列进行广度优先搜索
        i = 0
        while i < len(neighbors):
            neighbor_idx = neighbors[i]
            
            # 如果是噪声点，改为边界点
            if labels[neighbor_idx] == -1:
                labels[neighbor_idx] = cluster_id
            
            # 如果未访问过
            elif labels[neighbor_idx] == 0:
                labels[neighbor_idx] = cluster_id
                
                # 查询该点的邻域
                neighbor_neighbors = self._region_query(X, neighbor_idx)
                
                # 如果该点也是核心点，将其邻域加入队列
                if len(neighbor_neighbors) >= self.min_samples:
                    neighbors = np.concatenate([neighbors, neighbor_neighbors])
            
            i += 1
        
        return labels
    
    def fit_predict(self, X):
        """训练DBSCAN模型并返回聚类标签"""
        print(f"开始DBSCAN聚类，eps={self.eps}, min_samples={self.min_samples}")
        
        n_samples = X.shape[0]
        
        # 初始化标签：0表示未访问，-1表示噪声
        labels = np.zeros(n_samples, dtype=int)
        
        cluster_id = 0
        core_sample_indices = []
        
        # 遍历所有点
        for point_idx in range(n_samples):
            # 跳过已访问的点
            if labels[point_idx] != 0:
                continue
            
            # 查询邻域
            neighbors = self._region_query(X, point_idx)
            
            # 如果邻域内点数不足，标记为噪声
            if len(neighbors) < self.min_samples:
                labels[point_idx] = -1
            else:
                # 是核心点，创建新簇
                cluster_id += 1
                core_sample_indices.append(point_idx)
                labels = self._expand_cluster(X, labels, point_idx, neighbors, cluster_id)
            
            if point_idx % 1000 == 0:
                print(f"处理进度: {point_idx}/{n_samples}")
        
        self.labels = labels
        self.core_sample_indices = np.array(core_sample_indices)
        
        n_clusters = len(set(labels)) - (1 if -1 in labels else 0)
        n_noise = list(labels).count(-1)
        
        print(f"DBSCAN聚类完成！")
        print(f"发现簇数: {n_clusters}")
        print(f"噪声点数: {n_noise}")
        print(f"核心点数: {len(core_sample_indices)}")
        
        return self.labels
    
    def plot_clusters(self, X, y_true=None, save_path=None):
        """可视化聚类结果（使用PCA降维到2D）"""
        # 使用PCA降维到2D
        pca = PCA(n_components=2, random_state=42)
        X_2d = pca.fit_transform(X)
        
        # 创建图形
        fig, axes = plt.subplots(1, 2 if y_true is not None else 1, figsize=(16, 6))
        
        if y_true is not None:
            ax1 = axes[0]
        else:
            ax1 = axes if not isinstance(axes, np.ndarray) else axes[0]
        
        # 左图：DBSCAN聚类结果
        unique_labels = set(self.labels)
        n_clusters = len(unique_labels) - (1 if -1 in unique_labels else 0)
        
        colors = plt.cm.Spectral(np.linspace(0, 1, len(unique_labels)))
        
        for label, color in zip(unique_labels, colors):
            if label == -1:
                # 噪声点用黑色表示
                color = 'k'
                marker = 'x'
                label_name = '噪声'
            else:
                marker = 'o'
                label_name = f'簇{label}'
            
            class_member_mask = (self.labels == label)
            xy = X_2d[class_member_mask]
            ax1.scatter(xy[:, 0], xy[:, 1], c=[color], marker=marker,
                       s=50 if label == -1 else 30, label=label_name, 
                       alpha=0.6, edgecolors='k', linewidths=0.5)
        
        # 标记核心点
        if self.core_sample_indices is not None and len(self.core_sample_indices) > 0:
            ax1.scatter(X_2d[self.core_sample_indices, 0], 
                       X_2d[self.core_sample_indices, 1],
                       c='', marker='o', s=100, edgecolors='red', 
                       linewidths=2, label='核心点', alpha=0.3)
        
        n_noise = list(self.labels).count(-1)
        ax1.set_title(f'DBSCAN聚类结果\n簇数: {n_clusters}, 噪声点: {n_noise}', 
                     fontsize=14, fontweight='bold')
        ax1.set_xlabel('第一主成分', fontsize=12)
        ax1.set_ylabel('第二主成分', fontsize=12)
        ax1.legend(loc='best', fontsize=8)
        ax1.grid(True, alpha=0.3)
        
        if y_true is not None:
            # 右图：真实标签
            ax2 = axes[1]
            scatter2 = ax2.scatter(X_2d[:, 0], X_2d[:, 1], c=y_true, 
                                  cmap='tab10', alpha=0.6, s=30, 
                                  edgecolors='k', linewidths=0.5)
            ax2.set_title('真实标签分布', fontsize=14, fontweight='bold')
            ax2.set_xlabel('第一主成分', fontsize=12)
            ax2.set_ylabel('第二主成分', fontsize=12)
            ax2.grid(True, alpha=0.3)
            
            # 计算ARI和NMI评分（排除噪声点）
            mask = self.labels != -1
            if np.sum(mask) > 0:
                ari = adjusted_rand_score(y_true[mask], self.labels[mask])
                nmi = normalized_mutual_info_score(y_true[mask], self.labels[mask])
                fig.suptitle(f'ARI: {ari:.4f}, NMI: {nmi:.4f} (排除噪声点)',
```

### 2.3 算法原理说明

#### K-Means算法

**目标函数**：
$$
J = \sum_{k=1}^{K} \sum_{x_i \in C_k} ||x_i - \mu_k||^2
$$

**迭代步骤**：
1. **分配步骤**：将每个样本分配到最近的质心
   $$
   C_k = \{x_i : ||x_i - \mu_k||^2 \leq ||x_i - \mu_j||^2, \forall j\}
   $$

2. **更新步骤**：重新计算每个簇的质心
   $$
   \mu_k = \frac{1}{|C_k|} \sum_{x_i \in C_k} x_i
   $$

#### DBSCAN算法

**核心概念**：
- **ε-邻域**：$N_\varepsilon(x) = \{y \in D \mid dist(x,y) \leq \varepsilon\}$
- **核心对象**：$|N_\varepsilon(x)| \geq MinPts$
- **密度可达**：通过核心对象链接的点属于同一簇

### 3.3 评估指标

#### ARI (Adjusted Rand Index)
调整兰德指数，衡量聚类结果与真实标签的一致性：
$$
ARI = \frac{RI - E[RI]}{max(RI) - E[RI]}
$$
- 取值范围：[-1, 1]
- 1表示完全一致，0表示随机聚类，负值表示比随机更差

#### NMI (Normalized Mutual Information)
归一化互信息，衡量聚类结果包含的真实标签信息量：
$$
NMI = \frac{2 \times I(U;V)}{H(U) + H(V)}
$$
- 取值范围：[0, 1]
- 1表示完全一致，0表示独立

```python
class MyKNN:
    """
    K-近邻分类算法实现
    
    算法原理：
    对于测试样本x，找到训练集中与其最近的k个邻居，
    通过多数投票确定x的类别。
    
    数学公式：
    - 欧氏距离：d(x,y) = √(Σ(x_i - y_i)²)
    - 预测：ŷ = argmax Σ I(y_i = c)
              c    i∈N_k(x)
    """
    
    def __init__(self, k=5):
        self.k = k
        self.X_train = None
        self.y_train = None
    
    def fit(self, X, y):
        """存储训练数据"""
        self.X_train = X
        self.y_train = y
        print(f"KNN训练完成，存储了 {len(y)} 个训练样本")
        return self
    
    def _euclidean_distance(self, x1, x2):
        """计算欧氏距离"""
        return np.sqrt(np.sum((x1 - x2)**2, axis=1))
    
    def predict(self, X):
        """预测测试数据"""
        print(f"开始预测 {X.shape[0]} 个样本...")
        predictions = []
        
        for i, x in enumerate(X):
            # 计算到所有训练样本的距离
            distances = self._euclidean_distance(self.X_train, x)
            
            # 找到最近的k个邻居的索引
            k_indices = np.argsort(distances)[:self.k]
            
            # 获取k个邻居的标签
            k_nearest_labels = self.y_train[k_indices]
            
            # 多数投票
            most_common = np.bincount(k_nearest_labels).argmax()
            predictions.append(most_common)
            
            if (i + 1) % 500 == 0:
                print(f"  已预测 {i + 1}/{X.shape[0]} 个样本")
        
        print("预测完成！")
        return np.array(predictions)
```

```python
class MyGaussianNB:
    """
    高斯朴素贝叶斯分类器
    
    算法原理：
    假设特征服从高斯分布，基于贝叶斯定理进行分类
    
    数学公式：
    - 后验概率：P(y|x) ∝ P(y) × ∏P(x_i|y)
    - 高斯分布：P(x_i|y) = (1/√(2πσ²)) × exp(-(x_i-μ)²/(2σ²))
    """
    
    def __init__(self):
        self.classes = None
        self.class_prior = {}
        self.mean = {}
        self.var = {}
    
    def fit(self, X, y):
        """训练朴素贝叶斯模型"""
        self.classes = np.unique(y)
        n_samples, n_features = X.shape
        
        # 计算每个类别的先验概率、均值和方差
        for c in self.classes:
            X_c = X[y == c]
            self.class_prior[c] = len(X_c) / n_samples
            self.mean[c] = np.mean(X_c, axis=0)
            self.var[c] = np.var(X_c, axis=0) + 1e-9  # 加小数避免除零
        
        print(f"朴素贝叶斯训练完成！类别数: {len(self.classes)}")
        return self
    
    def _gaussian_probability(self, x, mean, var):
        """计算高斯概率密度"""
        eps = 1e-9
        coeff = 1.0 / np.sqrt(2.0 * np.pi * var + eps)
        exponent = np.exp(-(x - mean)**2 / (2 * var + eps))
        return coeff * exponent
    
    def predict(self, X):
        """预测测试数据"""
        predictions = []
        
        for x in X:
            posteriors = []
            
            # 计算每个类别的后验概率
            for c in self.classes:
                prior = np.log(self.class_prior[c])
                likelihood = np.sum(np.log(self._gaussian_probability(
                    x, self.mean[c], self.var[c])))
                posterior = prior + likelihood
                posteriors.append(posterior)
            
            # 选择后验概率最大的类别
            predictions.append(self.classes[np.argmax(posteriors)])
        
        return np.array(predictions)
```

```python
class MyDecisionTree:
    """
    决策树分类算法实现(CART)
    
    算法原理：
    使用Gini系数作为分裂标准，递归构建二叉树
    
    Gini系数：Gini = 1 - Σ(p_i)²
    信息增益：Gain = Gini_parent - Σ(n_i/n) × Gini_i
    """
    
    def __init__(self, max_depth=10, min_samples_split=2, min_samples_leaf=1):
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.min_samples_leaf = min_samples_leaf
        self.root = None
    
    def _gini(self, y):
        """计算Gini系数"""
        _, counts = np.unique(y, return_counts=True)
        probabilities = counts / len(y)
        return 1 - np.sum(probabilities ** 2)
    
    def _best_split(self, X, y):
        """寻找最佳分裂点"""
        n_samples, n_features = X.shape
        best_gain = 0
        best_split = None
        current_gini = self._gini(y)
        
        # 遍历每个特征
        for feature_idx in range(n_features):
            thresholds = np.unique(X[:, feature_idx])
            
            # 遍历每个阈值
            for threshold in thresholds:
                # 分裂数据
                left_mask = X[:, feature_idx] <= threshold
                right_mask = ~left_mask
                
                if np.sum(left_mask) < self.min_samples_leaf or \
                   np.sum(right_mask) < self.min_samples_leaf:
                    continue
                
                # 计算信息增益
                n_left, n_right = np.sum(left_mask), np.sum(right_mask)
                gini_left = self._gini(y[left_mask])
                gini_right = self._gini(y[right_mask])
                weighted_gini = (n_left * gini_left + n_right * gini_right) / n_samples
                gain = current_gini - weighted_gini
                
                if gain > best_gain:
                    best_gain = gain
                    best_split = {
                        'feature_idx': feature_idx,
                        'threshold': threshold,
                        'gain': gain
                    }
        
        return best_split
```

```python
class MyLinearSVM:
    """
    线性SVM分类算法(梯度下降法)
    
    算法原理：
    使用Hinge损失函数，通过梯度下降优化
    
    损失函数：L = C × Σmax(0, 1 - y_i(w·x_i + b)) + ||w||²/2
    梯度：∂L/∂w = w - C × Σy_i·x_i (对误分类点)
    """
    
    def __init__(self, C=1.0, learning_rate=0.001, n_iterations=1000):
        self.C = C
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.w = None
        self.b = None
    
    def _one_vs_rest_fit(self, X, y, class_label):
        """训练一个二分类SVM(one-vs-rest)"""
        y_binary = np.where(y == class_label, 1, -1)
        n_samples, n_features = X.shape
        
        # 初始化参数
        w = np.random.randn(n_features) * 0.01
        b = 0.0
        
        # 梯度下降优化
        for iteration in range(self.n_iterations):
            linear_output = np.dot(X, w) + b
            condition = y_binary * linear_output < 1
            
            # 计算梯度
            dw = w - self.C * np.dot(y_binary[condition], X[condition])
            db = -self.C * np.sum(y_binary[condition])
            
            # 更新参数
            w -= self.learning_rate * dw
            b -= self.learning_rate * db
        
        return w, b
```

```python
class MyPerceptron:
    """
    感知机算法(原始形式)
    
    算法原理：
    对于误分类点：w = w + η·y_i·x_i, b = b + η·y_i
    
    决策函数：f(x) = sign(w·x + b)
    """
    
    def __init__(self, learning_rate=0.01, n_iterations=1000):
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.w = None
        self.b = None
    
    def _one_vs_rest_fit(self, X, y, class_label):
        """训练一个二分类感知机"""
        y_binary = np.where(y == class_label, 1, -1)
        n_samples, n_features = X.shape
        
        w = np.zeros(n_features)
        b = 0.0
        
        # 迭代训练
        for iteration in range(self.n_iterations):
            errors = 0
            
            for i in range(n_samples):
                linear_output = np.dot(w, X[i]) + b
                
                # 如果分类错误，更新权重
                if y_binary[i] * linear_output <= 0:
                    w += self.learning_rate * y_binary[i] * X[i]
                    b += self.learning_rate * y_binary[i]
                    errors += 1
            
            if errors == 0:
                break
        
        return w, b

class MyPerceptronDual:
    """
    感知机算法(对偶形式)
    
    算法原理：
    使用Gram矩阵G[i,j] = x_i·x_j加速计算
    对于误分类点：α_i = α_i + η, b = b + η·y_i
    最终权重：w = Σ(α_i · y_i · x_i)
    """
    
    def __init__(self, learning_rate=1.0, n_iterations=1000):
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.alpha = None
        self.b = None
    
    def _compute_gram_matrix(self, X):
        """计算Gram矩阵"""
        return np.dot(X, X.T)
    
    def _one_vs_rest_fit(self, X, y, class_label):
        """训练对偶感知机"""
        y_binary = np.where(y == class_label, 1, -1)
        n_samples = X.shape[0]
        
        # 计算Gram矩阵
        gram = self._compute_gram_matrix(X)
        
        alpha = np.zeros(n_samples)
        b = 0.0
        
        for iteration in range(self.n_iterations):
            errors = 0
            
            for i in range(n_samples):
                # 使用对偶形式计算决策函数
                decision = np.sum(alpha * y_binary * gram[i]) + b
                
                if y_binary[i] * decision <= 0:
                    alpha[i] += self.learning_rate
                    b += self.learning_rate * y_binary[i]
                    errors += 1
            
            if errors == 0:
                break
        
        return alpha, b, X, y_binary
```

## 6. 选用合适评判指标对算法执行结果进行定量分析

### 6.1 聚类算法评估指标

| 指标 | 公式 | 含义 | 取值范围 |
|------|------|------|----------|
| ARI | $\frac{RI - E[RI]}{max(RI) - E[RI]}$ | 调整兰德指数 | [-1, 1] |
| NMI | $\frac{2 \times I(U;V)}{H(U) + H(V)}$ | 归一化互信息 | [0, 1] |
| 惯性 | $\sum_{k=1}^{K} \sum_{x \in C_k} \|x - \mu_k\|^2$ | 簇内平方和 | [0, +∞) |

### 6.2 分类算法评估指标

| 指标 | 公式 | 含义 | 适用场景 |
|------|------|------|----------|
| 准确率 | $\frac{TP+TN}{Total}$ | 整体正确率 | 样本均衡 |
| 精确率 | $\frac{TP}{TP+FP}$ | 预测为正的准确性 | 关注误报 |
| 召回率 | $\frac{TP}{TP+FN}$ | 实际为正的检出率 | 关注漏报 |
| F1分数 | $\frac{2 \times P \times R}{P+R}$ | 精确率和召回率的调和平均 | 综合评估 |

### 6.3 混淆矩阵

混淆矩阵是评估分类效果的直观工具：

```
              预测结果
            正类  负类
实际  正类   TP   FN
结果  负类   FP   TN
```

- **TP (True Positive)**：真正例，实际为正类预测为正类
- **TN (True Negative)**：真负例，实际为负类预测为负类
- **FP (False Positive)**：假正例，实际为负类预测为正类
- **FN (False Negative)**：假负例，实际为正类预测为负类
